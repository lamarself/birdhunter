import pygame import random import math # إعدادات الشاشة والألوان pygame.init() SCREEN_WIDTH = 800 SCREEN_HEIGHT = 600 screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("لعبة صياد الطيور والكلب الذكي") clock = pygame.time.Clock() WHITE = (255, 255, 255) GREEN = (34, 139, 34) BLUE = (135, 206, 235) BROWN = (139, 69, 19) RED = (255, 0, 0) BLACK = (0, 0, 0) YELLOW = (255, 215, 0) # العدادات hits = 0 misses = 0 font = pygame.font.SysFont("Arial", 24) # فئة الطائر class Bird: def __init__(self): self.reset() def reset(self): self.x = -50 self.y = random.randint(50, 300) self.speed = random.randint(3, 7) self.alive = True self.falling = False def update(self): if self.alive and not self.falling: self.x += self.speed if self.x > SCREEN_WIDTH: self.reset() elif self.falling: self.y += 8 # سرعة السقوط if self.y >= SCREEN_HEIGHT - 70: # سقط على العشب self.falling = False def draw(self): if self.alive or self.falling: # رسم طائر مبسط (جسم وجناح) pygame.draw.ellipse(screen, RED if self.falling else BROWN, (self.x, self.y, 40, 25)) pygame.draw.polygon(screen, WHITE, [(self.x+15, self.y), (self.x+25, self.y-15), (self.x+20, self.y)]) # فئة الكلب class Dog: def __init__(self): self.x = 200 self.y = SCREEN_HEIGHT - 80 self.speed = 5 self.state = "IDLE" # IDLE, FETCHING, RETURNING self.target_bird = None def update(self, bird, basket_x): if self.state == "IDLE" and bird.falling and bird.y >= SCREEN_HEIGHT - 85: self.state = "FETCHING" self.target_bird = bird if self.state == "FETCHING": if self.x < self.target_bird.x: self.x += self.speed elif self.x > self.target_bird.x: self.x -= self.speed # عند الوصول للطائر، يمسكه ويعود if abs(self.x - self.target_bird.x) < 10: self.target_bird.alive = False self.state = "RETURNING" if self.state == "RETURNING": if self.x > basket_x: self.x -= self.speed elif self.x < basket_x: self.x += self.speed # عند الوصول للسلة يضع الطائر ويعود لنقطة البداية if abs(self.x - basket_x) < 10: self.state = "IDLE" bird.reset() def draw(self): # رسم كلب مبسط pygame.draw.rect(screen, YELLOW, (self.x, self.y, 50, 30)) # الجسم pygame.draw.rect(screen, YELLOW, (self.x + 35, self.y - 15, 15, 15)) # الرأس # كائنات اللعبة bird = Bird() dog = Dog() hunter_x, hunter_y = 100, SCREEN_HEIGHT - 100 basket_x, basket_y = 50, SCREEN_HEIGHT - 80 # حلقة اللعبة الرئيسية running = True while running: screen.fill(BLUE) # السماء pygame.draw.rect(screen, GREEN, (0, SCREEN_HEIGHT - 60, SCREEN_WIDTH, 60)) # الحقل pygame.draw.rect(screen, BLACK, (basket_x, basket_y, 40, 30)) # السلة # أحداث اللعبة for event in pygame.event.get(): if event.type == pygame.QUIT: running = False if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1: # ضغطة ماوس يسار mouse_x, mouse_y = pygame.mouse.get_pos() # التحقق من إصابة الطائر if bird.alive and not bird.falling: # حساب المسافة بين الضغطة ومركز الطائر distance = math.hypot((bird.x + 20) - mouse_x, (bird.y + 12) - mouse_y) if distance < 30: # الطائر أصيب bird.falling = True hits += 1 else: misses += 1 else: misses += 1 # تحديث الكائنات bird.update() dog.update(bird, basket_x) # رسم الصياد والبندقية (تتجه نحو الماوس) mouse_x, mouse_y = pygame.mouse.get_pos() angle = math.atan2(mouse_y - hunter_y, mouse_x - hunter_x) gun_end_x = hunter_x + 40 * math.cos(angle) gun_end_y = hunter_y + 40 * math.sin(angle) pygame.draw.circle(screen, BLACK, (hunter_x, hunter_y), 20) # الصياد pygame.draw.line(screen, BROWN, (hunter_x, hunter_y), (gun_end_x, gun_end_y), 5) # البندقية # رسم الطائر والكلب bird.draw() dog.draw() # عرض العدادات hit_text = font.render(f"Hits (Caught): {hits}", True, BLACK) miss_text = font.render(f"Misses: {misses}", True, BLACK) screen.blit(hit_text, (20, 20)) screen.blit(miss_text, (20, 50)) pygame.display.flip() clock.tick(60) pygame.quit()
